# 机房预约管理系统 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 构建前后端分离的机房预约管理系统（机房/机位/节次/预约+冲突检测/审批/排课/占用日历/仪表盘），用 Docker Compose 部署，替换 cgx 旧项目。

**Architecture:** Vue3+Vite SPA（Nginx 托管并反代 `/api`）调用 Django3.2+DRF 后端；MySQL5.7 存数据，Redis 做缓存与 JWT 黑名单。后端按领域拆 app（accounts/rooms/bookings/schedules/stats），前端按 views/stores/api 模块化。

**Tech Stack:** Python 3.10.11 / Django 3.2.25 / DRF 3.14 / SimpleJWT / MySQL 5.7.44 / Redis；Node 20 / Vue 3.4.38 / Vite 5.4.14 / Element Plus / FullCalendar / Pinia / axios；Docker Compose。

from datetime import date, time
import pytest
from django.test import TestCase
from apps.rooms.models import Room, Seat
from apps.bookings.models import Booking
from apps.bookings.conflicts import has_conflict

class ConflictTests(TestCase):
    def setUp(self):
        self.room = Room.objects.create(name="301", location="A", capacity=50,
            open_time=time(8), close_time=time(22), status="open")
    def test_overlap_same_room_whole(self):
        Booking.objects.create(room=self.room, date=date(2026,6,12),
            start_time=time(10), end_time=time(12), purpose="x", status="approved")
        assert has_conflict(room=self.room, seat=None, d=date(2026,6,12),
            start=time(11), end=time(13)) is True
    def test_no_overlap_different_time(self):
        Booking.objects.create(room=self.room, date=date(2026,6,12),
            start_time=time(10), end_time=time(12), purpose="x", status="approved")
        assert has_conflict(room=self.room, seat=None, d=date(2026,6,12),
            start=time(13), end=time(14)) is False
    def test_rejected_does_not_conflict(self):
        Booking.objects.create(room=self.room, date=date(2026,6,12),
            start_time=time(10), end_time=time(12), purpose="x", status="rejected")
        assert has_conflict(room=self.room, seat=None, d=date(2026,6,12),
            start=time(11), end=time(13)) is False

```python
def _range(b, timeslots=None):
    if b.start_time and b.end_time:
        return b.start_time, b.end_time
    return b.timeslot.start_time, b.timeslot.end_time

def has_conflict(room, seat, d, start, end, exclude_id=None, status_in=("approved",)):
    from apps.bookings.models import Booking
    qs = Booking.objects.filter(room=room, date=d, status__in=status_in)
    if exclude_id:
        qs = qs.exclude(id=exclude_id)
    # 整间预约与该房任意机位互斥；指定机位只与同机位或整间冲突
    for b in qs.select_related("timeslot"):
        if seat and b.seat_id and b.seat_id != seat.id:
            continue
        bs, be = _range(b)
        if start < be and bs < end:   # 区间重叠
            return True
    return False
```
